def modify_file_content(filename, output_filename):
    try:
        # Open the input file for reading
        with open(filename, 'r') as file:
            content = file.read()

        # Modify the content (example: converting to uppercase)
        modified_content = content.upper()

        # Open the output file for writing
        with open(output_filename, 'w') as output_file:
            output_file.write(modified_content)

        print(f"Modified content saved to {output_filename}")

    except FileNotFoundError:
        print("Error: The specified file does not exist.")
    except IOError:
        print("Error: Could not read or write to the file.")


input_filename = input("Enter the name of the file to read: ")
output_filename = "modified_output.txt"

modify_file_content(input_filename, output_filename)
